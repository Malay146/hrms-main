# PeoplePay360 HRMS — requirements
#
# NOTE: This is a Node.js / Next.js project, not Python. Do NOT run
# `pip install -r requirements.txt`. Packages are installed with npm from
# package.json / package-lock.json. This file lists what you need on your
# machine and the exact package versions the app currently runs on.

# ---------------------------------------------------------------------------
# System requirements
# ---------------------------------------------------------------------------
node >= 20            # tested on 24.18.0
npm >= 10             # tested on 11.16.0
postgresql >= 16      # local install on default port 5432 (tested on 18.6) — not Docker
redis                 # optional: shared rate-limit store (REDIS_URL); in-memory fallback otherwise

# ---------------------------------------------------------------------------
# Setup
# ---------------------------------------------------------------------------
# 1. Install PostgreSQL 16+ and start it on port 5432
# 2. Create the database "PeoplePay360" and a role matching DATABASE_URL
# 3. cp .env.example .env   and fill in real values (see "Environment variables")
# 4. npm install            (also runs `prisma generate`)
# 5. npx prisma migrate deploy
# 6. npm run db:seed
# 7. npm run dev            then open http://localhost:3000
#
# Run tests with: npm test

# ---------------------------------------------------------------------------
# Environment variables (.env)
# ---------------------------------------------------------------------------
# Required:  DATABASE_URL, BETTER_AUTH_SECRET, BETTER_AUTH_URL, NEXT_PUBLIC_APP_URL
# Email:     SMTP_HOST, SMTP_PORT, SMTP_USER, SMTP_PASS (Gmail App Password), SMTP_FROM
# Optional:  OPENAI_API_KEY (AI analytics / copilot), REDIS_URL, CRON_SECRET
